/*
 * Tandas · servidor (Cloudflare Worker)
 *  1) IA para invitados: guarda la clave de Anthropic en secreto y solo deja pasar a quien tenga código.
 *  2) Recordatorios de comidas por notificación push (Android e iPhone), aunque la app esté cerrada.
 *
 * Configuración en Cloudflare (Settings):
 *   Variables and Secrets
 *     ANTHROPIC_API_KEY   (Secret) clave sk-ant-…
 *     CODIGOS             (Secret) "nombre:codigo,nombre:codigo,…"
 *     VAPID_PRIVATE       (Secret) clave privada de notificaciones (JSON que te pasé)
 *     VAPID_PUBLIC        (Text)   clave pública de notificaciones
 *   Bindings → KV namespace, nombre de variable: USO
 *   Triggers → Cron Triggers: *\/15 * * * *   (cada 15 minutos)
 */
const ORIGEN = 'https://sebastianmeythaler-jpg.github.io';
const CONTACTO = 'https://sebastianmeythaler-jpg.github.io/tandas/';
const LIMITE_DIARIO = 60;
const MODELOS = ['claude-haiku-5-5', 'claude-sonnet-5-5'];
const MAX_TOKENS = 6000; // un ticket largo necesita 2.000-4.000 tokens de salida; con 2.000 se cortaba
const VENTANA_MIN = 180; // si el móvil estaba apagado, el aviso aún se manda hasta 3 h después
const AVISOS = {
  desayuno: 'No te olvides de la comida más importante del día, prepárate tu desayuno',
  almuerzo: 'No te olvides de almorzar, es importante para tu salud',
  cena: 'No te olvides de cenar, también es importante para que duermas bien!'
};

/* ---------- utilidades ---------- */
const enc = new TextEncoder();
const b64u = buf => btoa(String.fromCharCode(...new Uint8Array(buf))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
const unb64u = s => { s = s.replace(/-/g, '+').replace(/_/g, '/'); while (s.length % 4) s += '='; return Uint8Array.from(atob(s), c => c.charCodeAt(0)); };
const concat = (...a) => { const out = new Uint8Array(a.reduce((n, x) => n + x.length, 0)); let o = 0; for (const x of a) { out.set(x, o); o += x.length; } return out; };
async function sha256hex(s) { return [...new Uint8Array(await crypto.subtle.digest('SHA-256', enc.encode(s)))].map(b => b.toString(16).padStart(2, '0')).join(''); }
function leerCodigos(txt) {
  const m = {};
  for (const par of String(txt || '').split(',')) { const [n, c] = par.split(':').map(s => (s || '').trim()); if (n && c) m[c.toLowerCase()] = n; }
  return m;
}
function horaLocal(tz, fecha = new Date()) {
  let p;
  try { p = new Intl.DateTimeFormat('en-GB', { timeZone: tz, year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', hourCycle: 'h23' }).formatToParts(fecha); }
  catch (e) { return horaLocal('Europe/Madrid', fecha); }
  const g = t => p.find(x => x.type === t).value;
  return { dia: `${g('year')}-${g('month')}-${g('day')}`, min: Number(g('hour')) * 60 + Number(g('minute')) };
}

/* ---------- Web Push (RFC 8291 + VAPID RFC 8292) ---------- */
async function hkdf(salt, ikm, info, len) {
  const k = await crypto.subtle.importKey('raw', ikm, 'HKDF', false, ['deriveBits']);
  return new Uint8Array(await crypto.subtle.deriveBits({ name: 'HKDF', hash: 'SHA-256', salt, info }, k, len * 8));
}
async function cifrar(payload, p256dh, auth) {
  const ua = unb64u(p256dh), secreto = unb64u(auth);
  const as = await crypto.subtle.generateKey({ name: 'ECDH', namedCurve: 'P-256' }, true, ['deriveBits']);
  const asPub = new Uint8Array(await crypto.subtle.exportKey('raw', as.publicKey));
  const uaKey = await crypto.subtle.importKey('raw', ua, { name: 'ECDH', namedCurve: 'P-256' }, false, []);
  const ecdh = new Uint8Array(await crypto.subtle.deriveBits({ name: 'ECDH', public: uaKey }, as.privateKey, 256));
  const ikm = await hkdf(secreto, ecdh, concat(enc.encode('WebPush: info\0'), ua, asPub), 32);
  const salt = crypto.getRandomValues(new Uint8Array(16));
  const cek = await hkdf(salt, ikm, enc.encode('Content-Encoding: aes128gcm\0'), 16);
  const nonce = await hkdf(salt, ikm, enc.encode('Content-Encoding: nonce\0'), 12);
  const key = await crypto.subtle.importKey('raw', cek, 'AES-GCM', false, ['encrypt']);
  const ct = new Uint8Array(await crypto.subtle.encrypt({ name: 'AES-GCM', iv: nonce }, key, concat(enc.encode(payload), new Uint8Array([2]))));
  const rs = new Uint8Array([0, 0, 0x10, 0]); // 4096
  return concat(salt, rs, new Uint8Array([asPub.length]), asPub, ct);
}
async function vapid(endpoint, env) {
  const jwk = JSON.parse(env.VAPID_PRIVATE);
  const key = await crypto.subtle.importKey('jwk', jwk, { name: 'ECDSA', namedCurve: 'P-256' }, false, ['sign']);
  const head = b64u(enc.encode(JSON.stringify({ typ: 'JWT', alg: 'ES256' })));
  const body = b64u(enc.encode(JSON.stringify({ aud: new URL(endpoint).origin, exp: Math.floor(Date.now() / 1000) + 12 * 3600, sub: CONTACTO })));
  const sig = await crypto.subtle.sign({ name: 'ECDSA', hash: 'SHA-256' }, key, enc.encode(`${head}.${body}`));
  return `vapid t=${head}.${body}.${b64u(sig)}, k=${env.VAPID_PUBLIC}`;
}
async function enviarPush(sub, datos, env) {
  const body = await cifrar(JSON.stringify(datos), sub.k.p256dh, sub.k.auth);
  return fetch(sub.e, { method: 'POST', headers: { 'Content-Encoding': 'aes128gcm', 'Content-Type': 'application/octet-stream', TTL: '10800', Urgency: 'high', Authorization: await vapid(sub.e, env) }, body });
}

/* ---------- recordatorios ---------- */
async function guardarSub(env, id, rec) { await env.USO.put('sub:' + id, JSON.stringify(rec), { metadata: rec }); }
async function revisarAvisos(env, ahora = new Date()) {
  let cursor; const enviados = [];
  do {
    const l = await env.USO.list({ prefix: 'sub:', cursor });
    for (const k of l.keys) {
      const rec = k.metadata || JSON.parse(await env.USO.get(k.name) || 'null'); if (!rec || !rec.on) continue;
      const { dia, min } = horaLocal(rec.tz, ahora); let cambio = false;
      for (const comida of Object.keys(AVISOS)) {
        const [h, m] = String((rec.t || {})[comida] || '').split(':').map(Number); if (isNaN(h)) continue;
        const desde = min - (h * 60 + (m || 0));
        if (desde < 0 || desde >= VENTANA_MIN) continue;
        if ((rec.done || {})[comida] === dia || (rec.sent || {})[comida] === dia) continue;
        rec.sent = rec.sent || {}; rec.sent[comida] = dia; cambio = true;
        const r = await enviarPush(rec, { title: 'Tandas', body: AVISOS[comida], tag: 'tandas-' + comida }, env);
        enviados.push([k.name, comida, r.status]);
        if (r.status === 404 || r.status === 410) { await env.USO.delete(k.name); cambio = false; break; }
      }
      if (cambio) await guardarSub(env, k.name.slice(4), rec);
    }
    cursor = l.list_complete ? null : l.cursor;
  } while (cursor);
  return enviados;
}

/* ---------- peticiones ---------- */
export default {
  async fetch(req, env) {
    const cors = { 'Access-Control-Allow-Origin': ORIGEN, 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'content-type, x-tandas-code', 'Access-Control-Max-Age': '86400', 'Vary': 'Origin' };
    const responder = (status, obj) => new Response(JSON.stringify(obj), { status, headers: { ...cors, 'content-type': 'application/json' } });
    const error = (status, tipo, mensaje) => responder(status, { type: 'error', error: { type: tipo, message: mensaje } });
    if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: cors });
    if (req.method !== 'POST') return error(405, 'metodo', 'Solo se aceptan peticiones POST.');
    const ruta = new URL(req.url).pathname.replace(/\/+$/, '') || '/';
    let body; try { body = await req.json(); } catch (e) { return error(400, 'peticion', 'Petición no válida.'); }

    // Marcar comida hecha: solo necesita el id secreto de la suscripción
    if (ruta === '/push/comi' || ruta === '/push/baja') {
      if (!env.USO || !/^[0-9a-f]{64}$/.test(body.id || '')) return error(400, 'peticion', 'Falta el id.');
      const rec = JSON.parse(await env.USO.get('sub:' + body.id) || 'null'); if (!rec) return responder(200, { ok: false });
      if (ruta === '/push/baja') { await env.USO.delete('sub:' + body.id); return responder(200, { ok: true }); }
      if (!AVISOS[body.comida]) return responder(200, { ok: true });
      rec.done = rec.done || {}; rec.done[body.comida] = horaLocal(rec.tz).dia; await guardarSub(env, body.id, rec);
      return responder(200, { ok: true });
    }

    // El resto necesita código de invitación
    const codigo = (req.headers.get('x-tandas-code') || '').trim().toLowerCase();
    const quien = leerCodigos(env.CODIGOS)[codigo];
    if (!quien) return error(401, 'codigo', 'Código de invitación no válido.');

    // Aprendizaje entre usuarios del mismo país: precios y nombres que salen de los tickets (sin datos personales)
    if (ruta === '/comunidad/aprende' || ruta === '/comunidad/leer') {
      if (!env.USO) return error(500, 'config', 'Falta el almacén USO.');
      const reg = String(body.region || '').toUpperCase().replace(/[^A-Z]/g, '').slice(0, 3) || 'ES';
      const clave = 'com:' + reg; const db = JSON.parse(await env.USO.get(clave) || '{"p":{},"n":{}}');
      if (ruta === '/comunidad/leer') {
        const precios = {}, nombres = {};
        for (const [k, v] of Object.entries(db.p)) if (v.c >= 1) precios[k] = Math.round(v.m * 10000) / 10000;
        for (const [k, v] of Object.entries(db.n)) { const top = Object.entries(v).sort((a, b) => b[1] - a[1])[0]; if (top) nombres[k] = top[0]; }
        return responder(200, { ok: true, precios, nombres });
      }
      for (const it of (body.items || []).slice(0, 80)) {
        const prod = String(it.p || '').slice(0, 60); if (!prod) continue;
        const pu = Number(it.pu); // precio por unidad del catálogo (g, ml o ud)
        if (pu > 0 && pu < 1000) { const v = db.p[prod] || { m: pu, c: 0 }; const w = Math.min(v.c, 9); v.m = (v.m * w + pu) / (w + 1); v.c = Math.min(v.c + 1, 999); db.p[prod] = v; }
        const nom = String(it.n || '').trim().slice(0, 40);
        if (nom) { const v = db.n[prod] || {}; v[nom] = (v[nom] || 0) + 1; db.n[prod] = v; }
      }
      await env.USO.put(clave, JSON.stringify(db));
      return responder(200, { ok: true });
    }

    if (ruta === '/push/alta') {
      if (!env.USO || !env.VAPID_PRIVATE || !env.VAPID_PUBLIC) return error(500, 'config', 'Faltan las claves de notificaciones en el servidor.');
      const s = body.sub || {}; if (!s.endpoint || !s.keys || !s.keys.p256dh || !s.keys.auth || !/^https:\/\//.test(s.endpoint)) return error(400, 'peticion', 'Suscripción no válida.');
      const id = await sha256hex(s.endpoint);
      const viejo = JSON.parse(await env.USO.get('sub:' + id) || 'null') || {};
      const t = {}; for (const c of Object.keys(AVISOS)) t[c] = /^\d{2}:\d{2}$/.test((body.t || {})[c] || '') ? body.t[c] : null;
      await guardarSub(env, id, { e: s.endpoint, k: { p256dh: s.keys.p256dh, auth: s.keys.auth }, tz: String(body.tz || 'Europe/Madrid').slice(0, 40), t, on: body.on !== false, done: { ...(viejo.done || {}), ...(body.done || {}) }, sent: viejo.sent || {}, q: quien });
      return responder(200, { ok: true, id });
    }

    if (ruta === '/push/prueba') {
      const id = body.id; const rec = /^[0-9a-f]{64}$/.test(id || '') && JSON.parse(await env.USO.get('sub:' + id) || 'null');
      if (!rec) return error(404, 'peticion', 'No encuentro tu suscripción.');
      const r = await enviarPush(rec, { title: 'Tandas', body: 'Así te llegarán los recordatorios. ¡Que aproveche!', tag: 'tandas-prueba' }, env);
      return responder(200, { ok: r.ok, status: r.status });
    }

    if (ruta !== '/' && ruta !== '/ia') return error(404, 'ruta', 'Ruta desconocida.');
    // IA
    if (!env.ANTHROPIC_API_KEY) return error(500, 'config', 'Falta configurar ANTHROPIC_API_KEY en el servidor.');
    const dia = new Date().toISOString().slice(0, 10), clave = `${dia}:${codigo}`; let usadas = 0;
    if (env.USO) { usadas = Number(await env.USO.get(clave)) || 0; if (usadas >= LIMITE_DIARIO) return error(429, 'limite', 'Límite diario alcanzado.'); }
    if (!MODELOS.includes(body.model)) body.model = MODELOS[0];
    body.max_tokens = Math.min(Number(body.max_tokens) || 1000, MAX_TOKENS);
    // Pensamiento apagado siempre (lo fija el servidor para que también valga con versiones antiguas de la app): Haiku «disabled», Sonnet «between_tools».
    body.thinking = body.model === 'claude-haiku-5-5' ? { type: 'disabled' } : { type: 'between_tools' };
    delete body.stream; body.metadata = { user_id: 'tandas-' + quien };
    const r = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'content-type': 'application/json', 'x-api-key': env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01' }, body: JSON.stringify(body) });
    if (env.USO && r.ok) await env.USO.put(clave, String(usadas + 1), { expirationTtl: 60 * 60 * 48 });
    return new Response(r.body, { status: r.status, headers: { ...cors, 'content-type': 'application/json' } });
  },

  async scheduled(event, env, ctx) { if (env.USO && env.VAPID_PRIVATE) ctx.waitUntil(revisarAvisos(env)); }
};
